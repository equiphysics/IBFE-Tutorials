% ---------------- Fingerprint ----------------
\begin{frame}{A practical ``fingerprint'' of turbulence}
\textbf{We will treat turbulence as flows that are typically:}
\begin{enumerate}
  \item \textbf{Three-dimensional} (3D vortical motion),
  \item \textbf{Multi-scale} in space and time (big eddies $\rightarrow$ small eddies),
  \item \textbf{Unsteady} (time-evolving),
  \item \textbf{Mixing} and strongly transporting scalar/particles.
\end{enumerate}

\vspace{0.7em}
\textbf{Mental experiment:} To characterize a turbulent flow, you can add dye upstream$\Rightarrow$ rapid stretching + folding $\Rightarrow$ fast mixing.

\end{frame}

% ---------------- 2D caveat ----------------
\begin{frame}{A note on 2D vs 3D turbulence}
\begin{itemize}
  \item Most ``everyday'' turbulence is \textbf{3D}.
  \item Some systems are \textbf{approximately 2D} (thin atmosphere/ocean surface).
  \item 2D turbulence can show different behavior (e.g., organization into large vortices).
\end{itemize}

% ---------------- Rayleigh-Bénard + chaos ----------------
\begin{frame}{Rayleigh--B\'enard convection and chaos}
\begin{itemize}

  \item Unsteady + multi-scale + mixing (often used to motivate chaotic dynamics).

% ---------------- Navier–Stokes entry ----------------
\begin{frame}{Mathematical entry point: incompressible Navier--Stokes}
\begin{align*}
\frac{\partial\uu}{\partial t} + (\uu\cdot\nabla)\uu &= -\nabla \pp + \frac{1}{Re}\, \Delta \uu,\\
\nabla\cdot\uu &= 0.
\end{align*}

\[
\mathbf{u}(\mathbf{x},t)=
\begin{bmatrix}
u(\mathbf{x},t)\\
v(\mathbf{x},t)\\
w(\mathbf{x},t)
\end{bmatrix}
\]


\begin{itemize}
  \item $\uu(\bm{x},t)\in\R^3$ is a 3-component velocity field.
  \item Rich behavior emerges as parameters/forcing/geometry change.
\end{itemize}

\end{frame}

% ---------------- Reynolds number ----------------
\begin{frame}{Reynolds number: a control knob for complexity}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.58\textwidth}
    \begin{itemize}
      \item The \textbf{Reynolds number} is a dimensionless ratio of inertial to viscous effects:
      \[
        \Rey \sim \frac{U L}{\nuu}.
      \]
      \item Increasing $U$ or $L$ generally increases flow complexity and turbulence.
      \item Rule of thumb from lecture: $\Rey$ relates large scales to smallest scales present.
    \end{itemize}
  \end{column}

  \begin{column}{0.42\textwidth}
    \centering
    \includegraphics[width=0.8\linewidth]{pictures/5-1/flow-cylinder.png}
  \end{column}
\end{columns}
\end{frame}

\begin{itemize}
  \item As flow regime changes with $\Rey$, the drag coefficient can change dramatically.
  \item Implication: ``just changing speed'' can change force laws in big ways.
\end{itemize}

% ---------------- Osborne Reynolds ----------------
\begin{frame}{Why ``Reynolds''? Pipe-flow experiments (1883)}
\begin{itemize}
  \item Osborne Reynolds observed transitions in pipe flow as speed increased:
  \begin{itemize}
    \item laminar (layered, steady),
    \item transitional,
    \item turbulent (roiling, disordered).
  \end{itemize}
  \item Pipe flow remains a core engineering problem: pumping water/sewage/fluids efficiently.
\end{itemize}

\begin{frame}{Transition to turbulence}
\begin{itemize}
  \item A classical experiment is to inject dye into a tube to visualize the flow.
  \item As the Reynolds number is increased (by increasing speed), the line of dye becomes unsteady and then turbulent.
\end{itemize}

\begin{frame}{Preview: Reynolds decomposition}

Recall the Navier-Stokes equations:

\begin{align*}
\frac{\partial\uu}{\partial t} + (\uu\cdot\nabla)\uu &= -\nabla \pp + \frac{1}{Re}\, \Delta \uu,\\
\nabla\cdot\uu &= 0.
\end{align*}

To model turbulence, we will use the Reynolds Decomposition:
\[
\uu(\bm{x},t) = \overline{\uu}(\bm{x}) + \uu'(\bm{x},t),
\qquad
\overline{\uu}(\bm{x}) := \lim_{T\to\infty}\frac{1}{T}\int_{0}^{T}\mathbf{u}(\mathbf{x},t)\,dt.
\]
\begin{itemize}
  \item Mean flow $\overline{\uu}$: time-averaged ``average frame of the movie''.
  \item Fluctuations $\uu'$: turbulent deviations around the mean.
\end{itemize}

\end{frame}

\begin{frame}{Why averaging became natural historically}
\begin{itemize}
  \item Before modern field measurements (e.g., PIV/laser sheets), data were often pointwise.
  \item Hot-wire anemometry: long time series at one point $\rightarrow$ build statistics.
  \item Averaging is both practical (data) and useful (reduced modeling complexity).

\end{itemize}

\end{frame}

\begin{frame}{Takeaways}
\begin{itemize}
  \item Turbulence is hard to define, but easy to recognize by its \textbf{fingerprint}:
  \[
  \text{3D} \;+\; \text{multi-scale} \;+\; \text{unsteady} \;+\; \text{mixing}.
  \]
  \item The Navier--Stokes equations remain the anchor; $\Rey$ controls regime/complexity.
  \item Coherent structures and intermittency mean turbulence is \textbf{not} pure randomness.
  \item Canonical flows are how we build understanding and models.
\item What's next: Reynolds averaging/closure.
\end{itemize}
\end{frame}
